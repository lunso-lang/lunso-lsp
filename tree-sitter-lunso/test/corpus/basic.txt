import std.io
const MAX = 0x10

enum Kind : u8 {
    One = 'a',
    Two,
}

struct Pair {
    a: i32
    b: []u8
}

pub fun add(x: i32, y: i32): i32 {
    var p = Pair { a: x, b: "ok" }
    match Kind.One {
        Kind.One => return p.a + y
        Kind.Two => return 0
    }
}

pub struct Vec[T] {
    data: *T
    len: u64
}

pub fun push[T](v: *Vec[T], x: T): T {
    return x
}

fun main(): i32 {
    var nums = Vec[int] { data: 0 as *int, len: 0 }
    return push.[int](&nums, 42)
}
