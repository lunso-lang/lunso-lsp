const MAX = 0x10

enum Kind : u8 {
    One = 'a',
    Two,
}

struct Pair {
    a: i32
    b: []u8
}

pub fun add(x: i32, y: i32): i32 {
    var p = Pair { a: x, b: "ok" }
    match Kind.One {
        Kind.One => return p.a + y
        Kind.Two => return 0
    }
}
